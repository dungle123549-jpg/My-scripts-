-- ==================== UTILITIES ====================
local Util_Func = {}

function Util_Func.GetRoot(character)
    if not character then return nil end
    return character:FindFirstChild("HumanoidRootPart") or character.PrimaryPart
end

function Util_Func.GetHumanoid(character)
    if not character then return nil end
    return character:FindFirstChildOfClass("Humanoid")
end

function Util_Func.IsAlive(character)
    if not character or not character.Parent then return false end
    local h = Util_Func.GetHumanoid(character)
    return h and h.Health > 0
end

function Util_Func.IsReady(instance)
    if not instance or not instance.Parent then return false end
    if not instance:IsDescendantOf(Workspace) then return false end
    if instance.Parent == Boats then return false end
    if instance:GetAttribute("IsBoat") then return false end
    if instance.Name:find("Boat") or instance.Name:find("Ship") then return false end
    if instance.Parent == SeaBeasts then
        local hp = instance:FindFirstChild("Health")
        return hp and hp.Value > 0
    end
    local h = Util_Func.GetHumanoid(instance)
    local r = Util_Func.GetRoot(instance)
    return h and h.Health > 0 and r
end

function Util_Func.Distance(pos1, pos2)
    if typeof(pos1) == "Instance" then
        pos1 = Util_Func.GetRoot(pos1) and Util_Func.GetRoot(pos1).Position or pos1.Position
    elseif typeof(pos1) == "CFrame" then
        pos1 = pos1.Position
    end
    if typeof(pos2) == "Instance" then
        pos2 = Util_Func.GetRoot(pos2) and Util_Func.GetRoot(pos2).Position or pos2.Position
    elseif typeof(pos2) == "CFrame" then
        pos2 = pos2.Position
    end
    return (pos1 - pos2).Magnitude
end

function Util_Func.FireInvoke(...)
    return CommF:InvokeServer(...)
end

end

function Util_Func.GetEquippedTool()

end

function Util_Func.EquipTool(toolName)
    if not Backpack or not Humanoid then return end
    local tool = Backpack:FindFirstChild(toolName) or Character:FindFirstChild(toolName)
    if tool then
        Humanoid:EquipTool(tool)
        return true
    end
    return false
end

function Util_Func.EquipWeapon(weaponType)
    if not Backpack or not Humanoid then return end
    for _, tool in ipairs(Backpack:GetChildren()) do
        if tool:IsA("Tool") and (not weaponType or tool.ToolTip == weaponType) then
            Humanoid:EquipTool(tool)
            return true
        end
    end
    for _, tool in ipairs(Character:GetChildren()) do
        if tool:IsA("Tool") and (not weaponType or tool.ToolTip == weaponType) then
            return true
        end
    end
    return false
end

function Util_Func.GetQuestMob()
    local main = PlayerGui:FindFirstChild("Main")
    local quest = main and main:FindFirstChild("Quest")
    if not quest or not quest.Visible then return nil end
    local container = quest:FindFirstChild("Container")
    local title = container and container:FindFirstChild("QuestTitle") and container.QuestTitle:FindFirstChild("Title")
    if title and title.Text ~= "" then
        local mob = title.Text:match("Defeat%s+%d+%s+(.-)%s+%(") or title.Text:match("Defeat%s+(.-)%s+%(")
        if mob then return mob end
    end
    return nil
end

function Util_Func.CheckInventory(category, name)
    local ok, result = pcall(function()
        return Net:RemoteFunction("GetAllItemValues"):InvokeServer()
    end)
    if not ok or not result then return nil end
    local grouped = {}
    for _, v in next, result, nil do
        local key = tostring(v.ItemId) .. "_" .. tostring(v.NetworkedUID)
        grouped[key] = grouped[key] or {ItemId = v.ItemId, Attributes = {}}
        if v.Value ~= nil then
            grouped[key].Attributes[v.Key] = v.Value
        end
    end
    for _, item in next, grouped, nil do
        local ok2, res2 = pcall(function()
            return ItemConfig.match(item.ItemId)
        end)
        if ok2 and res2 and res2.isOk and res2:isOk() then
            local idx = res2:unwrap()
            if idx and idx.Index and idx.Index.DebugLabel then
                local label = tostring(idx.Index.DebugLabel)
                local match = label:match("^(.-)%s*%[") or label
                if match:lower() == name:lower() then
                    if category == "Material" then
                        return item.Attributes.Quantity or 0
                    elseif category == "Moveset" then
                        return item.Attributes.Mastery or 0
                    end
                end
            end
        end
    end
    return category == "Material" and 0 or nil
end

-- ==================== TWEEN MANAGER ====================
local TweenManager = {}
TweenManager._Active = false
TweenManager._Cancel = false

function TweenManager.Stop()
    TweenManager._Cancel = true
    TweenManager._Active = false
end

function TweenManager.Teleport(cframe, speed)
    if not Root or not Root.Parent then return end
    if typeof(cframe) == "Vector3" then cframe = CFrame.new(cframe) end
    TweenManager._Cancel = false
    TweenManager._Active = true
    local start = Root.CFrame
    local distance = (cframe.Position - start.Position).Magnitude
    if distance <= 5 then
        TweenManager._Active = false
        return
    end
    speed = speed or Config.TweenSpeed
    local startTime = tick()
    local duration = distance / speed
    while tick() - startTime < duration do
        if TweenManager._Cancel or not Root or not Root.Parent then break end
        if not Humanoid or Humanoid.Health <= 0 then break end
        local alpha = math.min((tick() - startTime) / duration, 1)
        Root.CFrame = start:Lerp(cframe, alpha)
        RunService.Heartbeat:Wait()
    end
    if Root and Root.Parent then
        Root.CFrame = cframe
    end
    TweenManager._Active = false
end

function TweenManager.TeleportTask(cframe, speed)
    task.spawn(function()
        TweenManager.Teleport(cframe, speed)
    end)
end

FastAttack.GetHits = function(radius)
    radius = radius or 500
    local hits = {}
    local myRoot = Util_Func.GetRoot(Character)
    if not myRoot then return hits end
    
    if Config.AttackMobs then
        for _, enemy in ipairs(Enemies:GetChildren()) do
            if Util_Func.IsReady(enemy) then
                local eRoot = Util_Func.GetRoot(enemy)
                if eRoot then
                    local dist = (eRoot.Position - myRoot.Position).Magnitude
                    if dist <= radius then
                        table.insert(hits, {enemy, eRoot, dist})
                    end
                end
            end
        end
    end
    
    if Config.AttackPlayers then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LP and Util_Func.IsAlive(player.Character) then
                if not (player.Team and LP.Team and player.Team == LP.Team) then
                    local pRoot = Util_Func.GetRoot(player.Character)
                    if pRoot then
                        local dist = (pRoot.Position - myRoot.Position).Magnitude
                        if dist <= radius then
                            table.insert(hits, {player.Character, pRoot, dist})
                        end
                    end
                end
            end
        end
    end
    
    table.sort(hits, function(a, b) return a[3] < b[3] end)
    local result = {}
    for i = 1, math.min(#hits, 5) do
        result[i] = {hits[i][1], hits[i][2]}
    end
    return result
end

FastAttack.Attack = function()
    if not Character or not Humanoid or Humanoid.Health <= 0 then return end
    local tool = Util_Func.GetEquippedTool()
    if not tool then
        Util_Func.EquipWeapon(Config.SelectWeapon)
        tool = Util_Func.GetEquippedTool()
        if not tool then return end
    end
    
    if tool.ToolTip == "Gun" then
        FastAttack.ShootGun()
        return
    end
    
    local hits = FastAttack.GetHits(500)
    if #hits == 0 then return end
    
    if reRegisterAttack then
        pcall(function()
            reRegisterAttack:FireServer(Config.FastAttackDelay or 0.3)
        end)
    end
    
    if tool:FindFirstChild("LeftClickRemote") then
        pcall(function()
            tool.LeftClickRemote:FireServer(Vector3.new(0.01, -500, 0.01), 1, true)
        end)
    end
end

FastAttack.ShootGun = function(target)
    local tool = Util_Func.GetEquippedTool()
    if not tool or tool.ToolTip ~= "Gun" then return end
    if not tool.Enabled then return end
    tool.Enabled = true
    tool:SetAttribute("IsReloading_Client", false)
    tool:SetAttribute("LocalShotsLeft", 100)
    
    local targetPos
    if target then
        targetPos = Util_Func.GetRoot(target).Position
    else
        local hits = FastAttack.GetHits(500)
        if #hits == 0 then return end
        targetPos = hits[1][2].Position
    end
    
    pcall(function()
        if tool:FindFirstChild("RemoteEvent") then
            tool.RemoteEvent:FireServer("TAP", targetPos)
        elseif reShootGunEvent then
            reShootGunEvent:FireServer(targetPos, {})
        end
    end)
end

-- ==================== ESP SYSTEM ====================
local ESP = {}
ESP._Cache = setmetatable({}, {__mode = "k"})

function ESP.Clear(category)
    for instance, data in pairs(ESP._Cache) do
        if not category or data.Category == category then
            if data.Billboard and data.Billboard.Parent then
                data.Billboard:Destroy()
            end
            ESP._Cache[instance] = nil
        end
    end
end

function ESP.Add(adornee, text, color, category)
    if not adornee or ESP._Cache[adornee] then return end
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "DungdxESP"
    billboard.Adornee = adornee
    billboard.Size = UDim2.new(0, 200, 0, 50)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.MaxDistance = 5000
    billboard.Parent = adornee
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.TextColor3 = color or Color3.fromRGB(255, 255, 255)
    label.TextStrokeTransparency = 0.3
    label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    label.TextSize = 13
    label.Font = Enum.Font.GothamBold
    label.RichText = true
    label.Text = text or ""
    label.Parent = billboard
    
    ESP._Cache[adornee] = {Billboard = billboard, Category = category, Label = label}
end

function ESP.Update()
    if Config.ESPPlayer then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LP and Util_Func.IsAlive(player.Character) then
                local head = player.Character:FindFirstChild("Head")
                if head and not ESP._Cache[head] then
                    local level = player:FindFirstChild("Data") and player.Data:FindFirstChild("Level")
                    local lvl = level and level.Value or "?"
                    ESP.Add(head, string.format("<b>%s</b> [Lv.%s]", player.Name, lvl), Color3.fromRGB(255, 100, 100), "Player")
                end
            end
        end
    else
        ESP.Clear("Player")
    end
    
    if Config.ESPChest then
        local chestModels = Workspace:FindFirstChild("ChestModels")
        if chestModels then
            for _, chest in ipairs(chestModels:GetChildren()) do
                local root = chest:FindFirstChild("RootPart")
                if root and not ESP._Cache[root] then
                    local name = chest.Name:lower()
                    local color = Color3.fromRGB(255, 255, 255)
                    local label = "Chest"
                    if name:find("silver") then color = Color3.fromRGB(200, 200, 220); label = "Silver"
                    elseif name:find("gold") then color = Color3.fromRGB(255, 215, 0); label = "Gold"
                    elseif name:find("diamond") then color = Color3.fromRGB(0, 220, 255); label = "Diamond" end
                    ESP.Add(root, label, color, "Chest")
                end
            end
        end
    else
        ESP.Clear("Chest")
    end
    
    if Config.ESPFruit then
        for _, obj in ipairs(Workspace:GetChildren()) do
            if obj:IsA("Model") and obj.Name:find("Fruit") then
                local handle = obj:FindFirstChild("Handle") or obj.PrimaryPart
                if handle and not ESP._Cache[handle] then
                    ESP.Add(handle, obj.Name, Color3.fromRGB(255, 170, 0), "Fruit")
                end
            end
        end
    else
        ESP.Clear("Fruit")
    end
    
    if Config.ESPIsland then
        local locations = WorldOrigin and WorldOrigin:FindFirstChild("Locations")
        if locations then
            for _, loc in ipairs(locations:GetChildren()) do
                if loc:IsA("BasePart") and loc.Name ~= "Sea" and not ESP._Cache[loc] then
                    ESP.Add(loc, loc.Name, Color3.fromRGB(100, 200, 255), "Island")
                end
            end
        end
    else
        ESP.Clear("Island")
    end
end

local Functions = {}

Functions.AutoFarmLevel = function()
    while Config.AutoFarmLevel do
        task.wait(0.15)
        if not Util_Func.IsAlive(Character) then
            task.wait(1)
            continue
        end
        local mob = Util_Func.GetQuestMob()
        local hasQuest = Util_Func.HasQuest()
        local farmMode = Config.FarmMode
        
        if farmMode == "Quest" then
            if not hasQuest then
                if mob then
                    local npcs = ReplicatedStorage:FindFirstChild("NPCs")
                    if npcs then
                        for _, npc in ipairs(npcs:GetChildren()) do
                            if npc.Name:lower():find(mob:lower()) then
                                local npcRoot = Util_Func.GetRoot(npc) or npc:FindFirstChildWhichIsA("BasePart")
                                if npcRoot then
                                    TweenManager.Teleport(npcRoot.CFrame)
                                    task.wait(0.3)
                                    Util_Func.FireInvoke("StartQuest", "Quest", 1)
                                end
                                break
                            end
                        end
                    end
                end
                task.wait(0.5)
            else
                local found = false
                for _, enemy in ipairs(Enemies:GetChildren()) do
                    if Util_Func.IsReady(enemy) and enemy.Name:lower():find(mob:lower()) then
                        local eRoot = Util_Func.GetRoot(enemy)
                        if eRoot then
                            Util_Func.AutoHaki()
                            Util_Func.EquipWeapon(Config.SelectWeapon)
                            TweenManager.Teleport(eRoot.CFrame * CFrame.new(0, 15, 0))
                            if Config.BringMonster then
                                pcall(function()
                                    sethiddenproperty(LP, "SimulationRadius", Config.BringRadius)
                                end)
                            end
                            task.wait(0.05)
                            if FastAttack._Enabled then
                                FastAttack.Attack()
                            else
                                Humanoid:ChangeState(Enum.HumanoidStateType.Attack)
                            end
                            found = true
                        end
                        break
                    end
                end
                if not found then
                    local spawnFolder = ReplicatedStorage:FindFirstChild("FortBuilderReplicatedSpawnPositionsFolder")
                    if spawnFolder then
                        for _, part in ipairs(spawnFolder:GetChildren()) do
                            if part.Name == mob then
                                TweenManager.Teleport(part.CFrame + Vector3.new(0, 15, 0))
                                break
                            end
                        end
                    end
                end
            end
        elseif farmMode == "No Quest" or farmMode == "Nearest" then
            local nearest, minDist = nil, 500
            local myRoot = Util_Func.GetRoot(Character)
            for _, enemy in ipairs(Enemies:GetChildren()) do
                if Util_Func.IsReady(enemy) then
                    local eRoot = Util_Func.GetRoot(enemy)
                    if eRoot then
                        local dist = (eRoot.Position - myRoot.Position).Magnitude
                        if dist < minDist then
                            minDist = dist
                            nearest = enemy
                        end
                    end
                end
            end
            if nearest then
                local eRoot = Util_Func.GetRoot(nearest)
                Util_Func.AutoHaki()
                Util_Func.EquipWeapon(Config.SelectWeapon)
                TweenManager.Teleport(eRoot.CFrame * CFrame.new(0, 15, 0))
                task.wait(0.05)
                if FastAttack._Enabled then
                    FastAttack.Attack()
                else
                    Humanoid:ChangeState(Enum.HumanoidStateType.Attack)
                end
            end
        end
    end
end

-- ==================== WEBHOOK ====================
local Webhook = {}

Webhook.Send = function(data)
    if not Config.WebhookEnabled or Config.Webhook == "" then return end
    if not Config.Webhook:find("http") then return end
    task.spawn(function()
        pcall(function()
            local body = HttpService:JSONEncode(data)
            local req = request or http_request or (syn and syn.request)
            if req then
                req({
                    Url = Config.Webhook,
                    Method = "POST",
                    Headers = {["Content-Type"] = "application/json"},
                    Body = body
                })
            end
        end)
    end)
end

Webhook.SendSimple = function(title, content, color)
    Webhook.Send({
        embeds = {{
            title = title,
            description = content,
            color = color or 3066993,
            timestamp = DateTime.now():ToIsoDate(),
            footer = {text = "Blox Community VN | " .. LP.Name}
        }}
    })
end
